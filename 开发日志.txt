一、设置本机的虚拟机通过 IP 转发实现路由功能
1、虚拟机
# 开启 Linux 内核的 IPv4 转发功能，使虚拟机可以作为路由器转发网络数据
sudo sysctl -w net.ipv4.ip_forward=1

# 允许来自网口 enx9c69d36fe1c1（连接 RK3588） 的数据通过 ens33 转发到外部网络
sudo iptables -A FORWARD -i enx9c69d36fe1c1 -o ens33 -j ACCEPT

# 允许外部网络通过 ens33 返回的已建立连接数据转发回 RK3588
sudo iptables -A FORWARD -i ens33 -o enx9c69d36fe1c1 -m state --state RELATED,ESTABLISHED -j ACCEPT

# 对从 ens33 发出的数据进行 NAT 地址伪装，使 RK3588 可以借用虚拟机网络访问外网
sudo iptables -t nat -A POSTROUTING -o ens33 -j MASQUERADE

# 设置 DNS 服务器为 8.8.8.8，用于解析域名
